async function executePaystackPayment() { const reference = urlParams.get('reference') || localStorage.getItem('plh_pending_reference'); const country = urlParams.get('country'); if (!reference) { showToast('❌ Payment session expired. Please try again.', 'error'); return; } paymentInFlight = true; try { console.log('💳 Executing Paystack payment for:', { reference, country }); const { data, error } = await supabase.functions.invoke('paystack-initialize', { body: { amount: BOOKING_FEE_CENTS, email: currentUser.email, reference: reference, currency: CURRENCY, // Always KES callback_url: `${window.location.origin}${window.location.pathname}?conversation=${conversationId}&reference=${reference}`, metadata: { conversation_id: conversationId, user_id: currentUser.id, member_name: conversation.member_name || '', full_name: conversation.full_name || '', country: country } } }); console.log('=== PAYSTACK INITIALIZE RAW RESPONSE ==='); console.log('data:', JSON.stringify(data, null, 2)); console.log('error:', error); if (error) { console.error('Initialize error:', error); throw new Error( (data?.error) || (data?.message) || error.message || 'Failed to initialize payment' ); } if (!data || !data.authorization_url) { const diagnostic = data?.error ? `Function error: ${data.error}` : data?.message ? `Function message: ${data.message}` : `Raw response: ${JSON.stringify(data)}`; throw new Error(diagnostic); } // Update DB with pending status await supabase .from('conversations') .update({ payment_status: 'pending', payment_reference: reference }) .eq('id', conversationId); console.log('✅ Redirecting to Paystack checkout:', data.authorization_url); window.location.href = data.authorization_url; } catch (err) { console.error('Payment start failed:', err); showToast('❌ ' + (err.message || 'Payment failed. Please try again.'), 'error'); paymentInFlight = false; } }
